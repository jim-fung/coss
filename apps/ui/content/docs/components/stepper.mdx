## Usage

```tsx
import {
  Stepper,
  StepperIndicator,
  StepperItem,
  StepperSeparator,
  StepperTitle,
  StepperTrigger,
} from "@/components/ui/stepper"
```

```tsx
const [activeStep, setActiveStep] = useState(1)

<Stepper onValueChange={setActiveStep} value={activeStep}>
  {steps.map((step, index) => (
    <StepperItem completed={activeStep > index} key={step} step={index}>
      <StepperTrigger>
        <StepperIndicator />
        <StepperTitle>{step}</StepperTitle>
      </StepperTrigger>
      {index < steps.length - 1 && <StepperSeparator />}
    </StepperItem>
  ))}
</Stepper>
```

## Examples

### Vertical

Set `orientation="vertical"` on the `Stepper` to stack the steps.

### States

- `completed` — set `completed` on a `StepperItem` (or step below `value`) to show a checkmark.
- `loading` — set `loading` on the active `StepperItem` to show a spinner.
- `disabled` — set `disabled` on a `StepperItem` to make it non-interactive.
