## Usage

```tsx
import {
  ResizableHandle,
  ResizablePanel,
  ResizablePanelGroup,
} from "@/components/ui/resizable"
```

```tsx
<ResizablePanelGroup>
  <ResizablePanel defaultSize="30">One</ResizablePanel>
  <ResizableHandle />
  <ResizablePanel defaultSize="70">Two</ResizablePanel>
</ResizablePanelGroup>
```

## Examples

### Vertical

Set `orientation="vertical"` on the group.

### With handle

Add the `withHandle` prop to `ResizableHandle` to render a visible grip.
